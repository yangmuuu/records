uint bsp_get_bat_level(void)
{
#if VBAT_DETECT_EN
    u16 bat_off = sys_cb.vbat_pwroff;

    if (bat_off > sys_cb.vbat) {
        return 0;
    }

    uint bat_level =
        (sys_cb.vbat - bat_off) /
        ((4200 - bat_off) / 100);

    if (bat_level > 100) {
        bat_level = 100;
    }

    return bat_level;
#endif
}
